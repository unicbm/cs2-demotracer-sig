cmake_minimum_required(VERSION 3.20)
project(schema_probe CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT HL2SDK OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64 and -DHL2SDK=<SDK path>")
endif()
execute_process(COMMAND git -C "${HL2SDK}" rev-parse HEAD OUTPUT_VARIABLE SDK_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SDK_COMMIT STREQUAL "5f891c9026230cce0fc0a3fc4b5fef1c467a1385")
  message(FATAL_ERROR "Use alliedmodders/hl2sdk at 5f891c9026230cce0fc0a3fc4b5fef1c467a1385")
endif()
add_library(schema_probe SHARED probe.cpp)
target_include_directories(schema_probe PRIVATE ${HL2SDK}/public ${HL2SDK}/public/tier0 ${HL2SDK}/public/tier1 ${HL2SDK}/public/mathlib)
target_compile_definitions(schema_probe PRIVATE PLATFORM_64BITS X64BITS COMPILER_MSVC COMPILER_MSVC64 NOMINMAX _ALLOW_KEYWORD_MACROS)
target_link_libraries(schema_probe PRIVATE ${HL2SDK}/lib/public/win64/tier0.lib ${HL2SDK}/lib/public/win64/tier1.lib)
